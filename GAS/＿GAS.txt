/**
 * Drive から画像と下書きテキストを列挙して取得するモジュール
 * - スプレッドシートの A2 セルにフォルダIDを置く想定
 * - 画像は拡張子で判定（jpg|jpeg|png|webp）
 * - テキストは .txt/.md を取得して内容を読み取る
 * - 返り値は配列: { type: 'image'|'text'|'other', name, id, mime, content? }
 */

function getFolderFromSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var raw = sheet.getRange('A2').getValue();
  if (!raw) {
    throw new Error('フォルダIDまたはフォルダ名がシートの A2 に設定されていません');
  }
  var v = String(raw).trim();

  // 1) URL から ID を抜く（/folders/ID や id= など）
  var m = v.match(/[-\w]{20,}/);
  if (m) {
    var candidate = m[0];
    try {
      var folder = DriveApp.getFolderById(candidate);
      Logger.log('Folder found by ID: ' + folder.getName());
      return folder;
    } catch (e) {
      // 続行してフォルダ名検索へ
    }
  }

  // 2) そのまま ID として試す
  try {
    var folder = DriveApp.getFolderById(v);
    Logger.log('Folder found by direct ID: ' + folder.getName());
    return folder;
  } catch (e) {
    // 続行
  }

  // 3) フォルダ名として検索して最初に見つかったものを返す
  try {
    var folders = DriveApp.getFoldersByName(v);
    if (folders.hasNext()) {
      var folder = folders.next();
      Logger.log('Folder found by name: ' + folder.getName());
      return folder;
    }
  } catch (e) {
    // 最後にエラーを投げる
  }

  throw new Error('フォルダが見つかりません（A2: ' + v + '）。A2 にはフォルダのID、またはフォルダURL、または正確なフォルダ名を設定してください。');
}

function listMediaAndTexts(folder) {
  var files = folder.getFiles();
  var items = [];
  while (files.hasNext()) {
    var file = files.next();
    var name = file.getName();
    var mime = file.getMimeType();
    var id = file.getId();

    if (name.match(/\.(jpg|jpeg|png|webp)$/i)) {
      var lastUpdated = file.getLastUpdated ? file.getLastUpdated() : null;
      items.push({ type: 'image', name: name, id: id, mime: mime, lastUpdated: lastUpdated });
      continue;
    }

    if (name.match(/\.(txt|md)$/i) || mime === 'text/plain') {
      var content = '';
      try {
        content = file.getBlob().getDataAsString('utf-8');
      } catch (e) {
        content = file.getBlob().getDataAsString();
      }
      items.push({ type: 'text', name: name, id: id, mime: mime, content: content });
      continue;
    }

    items.push({ type: 'other', name: name, id: id, mime: mime });
  }
  return items;
}

/** 指定した items から最新画像をサムネイルとして選定して返す（なければ null） */
function selectLatestImage(items) {
  var images = items.filter(function(it) { return it.type === 'image'; });
  if (images.length === 0) return null;
  images.sort(function(a, b) {
    var at = a.lastUpdated ? new Date(a.lastUpdated).getTime() : 0;
    var bt = b.lastUpdated ? new Date(b.lastUpdated).getTime() : 0;
    return bt - at;
  });
  return images[0];
}

/**
 * 開発用エントリポイント: フォルダを読み取り、ログに出力し、簡易 manifest をフォルダへ書き出す
 */
function buildFolderManifest() {
  var folder = getFolderFromSheet();
  var items = listMediaAndTexts(folder);
  Logger.log('Found items: ' + items.length);
  items.forEach(function(it) {
    Logger.log(it.type + ': ' + it.name);
    if (it.type === 'text') {
      Logger.log('Text content: ' + it.content);
    }
  });

  // サムネイル選定（最新の画像をサムネイルとする）
  var thumbnail = selectLatestImage(items);
  if (thumbnail) {
    Logger.log('Selected thumbnail: ' + thumbnail.name);
  } else {
    Logger.log('No thumbnail found');
  }

  // デバッグ用に manifest を作成（GAS 権限でフォルダに書き込み）
  try {
    var manifestObj = {
      thumbnail: thumbnail ? { name: thumbnail.name, id: thumbnail.id } : null,
      items: items.map(function(it){ return { type: it.type, name: it.name, id: it.id, mime: it.mime, lastUpdated: it.lastUpdated, content: it.content || null }; })
    };
    var manifest = JSON.stringify(manifestObj, null, 2);
    var manifestBlob = Utilities.newBlob(manifest, 'application/json', folder.getName() + '_manifest.json');
    folder.createFile(manifestBlob);
  } catch (e) {
    Logger.log('manifest 作成中にエラー: ' + e);
  }

  return items;
}

// -----------------------------
// Gemini API 呼び出しラッパー
// -----------------------------
function getGeminiConfig() {
  var p = PropertiesService.getScriptProperties();
  return {
    apiKey: p.getProperty('GEMINI_API_KEY'),
    apiUrl: p.getProperty('GEMINI_API_URL'),
    model: p.getProperty('GEMINI_MODEL') || 'gemini',
    // 一言キャプション用の低トークン設定
    captionMaxTokens: Number(p.getProperty('GEMINI_CAPTION_MAXTOKENS') || 60),
    articleMaxTokens: Number(p.getProperty('GEMINI_ARTICLE_MAXTOKENS') || 800)
  };
}

function callGeminiRaw(promptText, options) {
  options = options || {};
  var cfg = getGeminiConfig();
  var apiUrl = cfg.apiUrl;
  if (!apiUrl) throw new Error('Script Properties に GEMINI_API_URL を設定してください');
  var apiKey = cfg.apiKey;

  var body = {
    prompt: promptText,
    model: options.model || cfg.model,
    max_tokens: options.maxTokens || cfg.captionMaxTokens,
    temperature: typeof options.temperature !== 'undefined' ? options.temperature : 0.2
  };

  var headers = {};
  if (apiKey) headers['Authorization'] = 'Bearer ' + apiKey;

  var payload = JSON.stringify(body);
  var params = {
    method: 'post',
    contentType: 'application/json',
    payload: payload,
    muteHttpExceptions: true,
    headers: headers
  };

  var resp = UrlFetchApp.fetch(apiUrl, params);
  var code = resp.getResponseCode();
  var text = resp.getContentText();
  if (code < 200 || code >= 300) {
    throw new Error('Gemini API エラー: ' + code + ' - ' + text);
  }
  return text;
}

function callGeminiForJson(promptText, options) {
  var raw = callGeminiRaw(promptText, options);
  // 期待される出力は純粋な JSON 文字列
  try {
    return JSON.parse(raw);
  } catch (e) {
    // ログ出力して生テキストを返す
    Logger.log('callGeminiForJson: JSON parse error, raw=' + raw);
    return { raw: raw };
  }
}

// -----------------------------
// 画像とテキストの紐付け（簡易）
// - 画像名と同じベース名のテキストファイルがあれば紐付け
// - なければ最初に見つかった text を使用
// -----------------------------
function pairImagesWithTexts(items) {
  var images = items.filter(function(it){ return it.type === 'image'; });
  var texts = items.filter(function(it){ return it.type === 'text'; });
  var pairs = [];

  images.forEach(function(img){
    var base = img.name.replace(/\.[^.]+$/, '');
    var matched = texts.find(function(t){ return t.name.replace(/\.[^.]+$/, '') === base; });
    if (matched) {
      pairs.push({ image: img, text: matched.content });
    } else if (texts.length > 0) {
      pairs.push({ image: img, text: texts[0].content });
    } else {
      pairs.push({ image: img, text: '' });
    }
  });
  return pairs;
}

// -----------------------------
// 店舗情報取得: スプレッドシート内の '店舗情報' シートから folderName をキーに検索
// シート構成（想定）: A: folderName/B:店舗名/C:住所/D:営業時間/E:定休日/F:電話/G:公式サイト
// -----------------------------
function getStoreInfoFromSheet(folderName) {
  if (!folderName) return null;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('店舗情報') || ss.getSheetByName('StoreDirectory');
  if (!sh) return null;
  var data = sh.getDataRange().getValues();
  // assume header row at row 1; iterate from row 2
  // try to parse folderName like: 店名_地域名_県名
  var parts = String(folderName).split(/[_\s]+/).map(function(s){ return String(s||'').trim(); });
  var storeCandidate = parts[0] || '';
  var regionCandidate = parts[1] || '';
  var prefectureCandidate = parts[2] || '';

  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var key = String(row[0] || '').trim();
    if (!key) continue;
    // match priority: exact folderName, exact key==storeCandidate, partial contains
    if (key === folderName || key === storeCandidate || key.indexOf(folderName) !== -1 || folderName.indexOf(key) !== -1 || key.indexOf(storeCandidate) !== -1) {
      return {
        key: key,
        name: row[1] || '',
        address: row[2] || '',
        hours: row[3] || '',
        closed: row[4] || '',
        phone: row[5] || '',
        website: row[6] || ''
      };
    }
  }
  // not found in sheet: return parsed suggestions from folder name so caller can still use them
  return {
    key: folderName,
    name: storeCandidate,
    region: regionCandidate,
    prefecture: prefectureCandidate,
    address: '',
    hours: '',
    closed: '',
    phone: '',
    website: ''
  };
}

// -----------------------------
// 各画像ごとに短い一言キャプションを生成（低トークン）
// 返り値: items に caption を追加した配列
// -----------------------------
function generateCaptionsForItems(items) {
  var cfg = getGeminiConfig();
  var pairs = pairImagesWithTexts(items);
  pairs.forEach(function(p, idx){
    var prompt = '以下の画像説明テキストに基づき、日本語で短い一言キャプション（1〜2文）と alt テキスト（短め）をJSONで返してください。' +
      '\n---\n' + (p.text || '(説明なし)') + '\n---\n' +
      '{"caption":"","alt":""} の形式で返してください。';
    try {
      var res = callGeminiForJson(prompt, { maxTokens: cfg.captionMaxTokens, temperature: 0.2 });
      if (res && res.caption) {
        p.image.caption = res.caption;
        p.image.alt = res.alt || res.caption;
      } else if (res.raw) {
        // 生テキストをキャプションにフォールバック
        p.image.caption = (typeof res.raw === 'string') ? res.raw.substring(0, 200) : '';
        p.image.alt = p.image.caption;
      }
    } catch (e) {
      Logger.log('キャプション生成エラー: ' + e);
      p.image.caption = '';
      p.image.alt = '';
    }
  });
  return items;
}

// -----------------------------
// 一言キャプションと下書きテキストを使って記事を生成
// 返り値: JSON { title, lead, bodyHtml, imageAlts: [{name,alt,caption}], tags: [] }
// -----------------------------
function generateArticleFromDraft(draftText, items) {
  var cfg = getGeminiConfig();
  var imageEntries = items.filter(function(it){ return it.type==='image'; }).map(function(it){
    return { name: it.name, caption: it.caption || '', alt: it.alt || '' };
  });
  // フォルダ名（= 想定される店舗名）を取得し、シートから店舗情報を探す（無ければ空オブジェクト）
  var folderName = null;
  try {
    folderName = getFolderFromSheet().getName();
  } catch (e) {
    folderName = null;
  }
  var storeInfo = {};
  try {
    if (folderName) {
      var si = getStoreInfoFromSheet(folderName);
      if (si) storeInfo = si;
    }
  } catch (e) {
    Logger.log('getStoreInfoFromSheet error: ' + e);
  }

  var prompt = 'あなたはブログ記事ライターです。以下の下書きと各画像の短いキャプション、そして店舗情報（あれば）を参照し、' +
    'ブログ記事のタイトル（短め）、リード文（導入文、約2〜3文）、本文（適切な見出しを含むHTML形式）、' +
    '各画像の alt テキスト、および店舗情報を JSON で返してください。出力は厳密な JSON としてください。\n\n' +
    '店舗情報は、本文の冒頭または適切な場所に以下の形式で記載してください：\n' +
    '■ 店舗情報\n' +
    '住所：' + (storeInfo.address || '') + '\n' +
    '営業時間：' + (storeInfo.hours || '') + '\n' +
    '定休日：' + (storeInfo.closed || '') + '\n' +
    '電話番号：' + (storeInfo.phone || '') + '\n' +
    '公式サイト：' + (storeInfo.website || '') + '\n\n' +
    '■ アクセス（地図）\n' +
    '住所からGoogle Mapsへのリンク：https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(storeInfo.address || '') + '\n\n' +
    '下書き:\n' + draftText + '\n\n' +
    '画像情報:\n' + JSON.stringify(imageEntries, null, 2) + '\n\n' +
    '店舗情報:\n' + JSON.stringify(storeInfo, null, 2) + '\n\n' +
    '期待する JSON フォーマットの例:\n' +
    '{"title":"...","lead":"...","bodyHtml":"<p>..</p>","imageAlts":[{"name":"...","alt":"...","caption":"..."}],"storeInfo":{"name":"","address":"","hours":"","closed":"","phone":"","website":""},"tags":["tag1","tag2"]}';

  try {
    var res = callGeminiForJson(prompt, { maxTokens: cfg.articleMaxTokens, temperature: 0.3 });
    return res;
  } catch (e) {
    Logger.log('記事生成エラー: ' + e);
    return { error: e.toString() };
  }
}

// テスト用: manifest をスプレッドシートに書き出す（手早く出力を確認できる）
function exportManifestToSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var items = buildFolderManifest();

  // ここでは A13 から一覧を書き出す（ユーザーの既存データ A1:A12 を上書きしない）
  sheet.getRange('A13:E10000').clearContent();
  sheet.getRange(14,1).setValue('type');
  sheet.getRange(14,2).setValue('name');
  sheet.getRange(14,3).setValue('id');
  sheet.getRange(14,4).setValue('mime');
  sheet.getRange(14,5).setValue('caption');

  for (var i = 0; i < items.length; i++) {
    var r = 15 + i;
    var it = items[i];
    sheet.getRange(r,1).setValue(it.type);
    sheet.getRange(r,2).setValue(it.name);
    sheet.getRange(r,3).setValue(it.id);
    sheet.getRange(r,4).setValue(it.mime);
    if (it.type === 'image' && it.caption) {
      sheet.getRange(r,5).setValue(it.caption);
    }
  }
  var thumbnail = selectLatestImage(items);
  // 状態表示は A13 に書き込む（A1:A12 は保持）
  sheet.getRange(13,1).setValue('manifest generated');
  sheet.getRange(13,2).setValue(thumbnail ? thumbnail.name : 'No thumbnail');
  Logger.log('exportManifestToSheet: wrote ' + items.length + ' items to sheet.');
  return items;
}

function generateMockCaptions(items) {
  var pairs = pairImagesWithTexts(items);
  pairs.forEach(function(p){
    var caption = '';
    if (p.text && p.text.trim().length > 0) {
      caption = p.text.split(/\r?\n/)[0];
      if (caption.length > 120) caption = caption.substring(0,120) + '...';
    } else {
      caption = '写真：' + p.image.name;
    }
    p.image.caption = caption;
    p.image.alt = caption;
  });
  return items;
}

// -----------------------------
// テキストファイルに地域情報を記入する関数
// - フォルダ名から地域情報を取得し、テキストファイルの先頭に追加
// -----------------------------
function appendRegionInfoToTexts(items, storeInfo) {
  var texts = items.filter(function(it) { return it.type === 'text'; });
  texts.forEach(function(textItem) {
    var originalContent = textItem.content || '';
    var regionInfo = '';
    if (storeInfo && storeInfo.region) {
      regionInfo = '地域: ' + storeInfo.region;
      if (storeInfo.prefecture) regionInfo += ' (' + storeInfo.prefecture + ')';
      regionInfo += '\n\n';
    }
    var newContent = regionInfo + originalContent;
    try {
      var file = DriveApp.getFileById(textItem.id);
      file.setContent(newContent);
      textItem.content = newContent; // items 配列も更新
      Logger.log('地域情報を追加: ' + textItem.name);
    } catch (e) {
      Logger.log('テキスト更新エラー (' + textItem.name + '): ' + e);
    }
  });
  return items;
}
